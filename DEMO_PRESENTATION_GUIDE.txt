Q: Bagaimana cara upload gambar?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
A: "Saat create atau edit produk, ada field 'Gambar Sepatu'.
   Klik Choose File, pilih gambar, preview langsung muncul.
   System validasi format (JPG/PNG/GIF/WEBP) dan ukuran (max 2MB).
   File disimpan di folder uploads/ dengan nama unik."


Q: Bagaimana keamanannya?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
A: "Kami menggunakan:
   1. PDO Prepared Statements - mencegah SQL Injection
   2. htmlspecialchars() - mencegah XSS
   3. File validation - hanya izinkan image files
   4. Input sanitization - clean semua input
   5. Server-side validation - tidak hanya client-side"


Q: Apa kegunaan Admin Panel?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
A: "Admin Panel memungkinkan:
   - Rebrand website tanpa edit code
   - Ganti warna sesuai brand
   - Toggle features on/off
   - Customize logo icon
   - Settings disimpan di database
   - Live preview sebelum save
   
   Ini membuat aplikasi lebih flexible dan reusable."


Q: Database structure seperti apa?
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
A: "2 tabel:
   
   Tabel 'sepatu':
   - 11 kolom (id, nama, merk, kategori, ukuran, warna,
     harga, stok, deskripsi, gambar, timestamps)
   - Primary key: id (auto increment)
   
   Tabel 'settings':
   - 4 kolom (id, setting_key, setting_value, updated_at)
   - Menyimpan konfigurasi website
   - Key-value pairs untuk flexibility"
