  ✓ Sanitasi data

  1. Terima data POST dari form

█████████████████████████████████████████████████████████████████████

1. CREATE (Tambah Data)
━━━━━━━━━━━━━━━━━━━━━━━
User Action:
  index.php → Klik "Tambah Produk" → Modal muncul → Isi form → Klik "Simpan"

Data Flow:
  Form (POST) → create.php → Validasi → Sanitasi → Database (INSERT)
                                    ↓
  index.php ← Redirect dengan message ← Success/Error


2. READ (Tampil Data)
━━━━━━━━━━━━━━━━━━━━━━━
User Action:
  Browser → http://localhost/starsports → index.php load

Data Flow:
  index.php → getAllSepatu() → Database (SELECT) → Fetch results
                                    ↓
  Render HTML table ← Loop through results ← PDO fetchAll()


3. UPDATE (Edit Data)
━━━━━━━━━━━━━━━━━━━━━━━
User Action:
  index.php → Klik tombol "Edit" → edit.php load → Ubah data → Klik "Update"

Data Flow:
  Step 1: Load form
    edit.php?id=X → Database (SELECT WHERE id=X) → Pre-fill form
  
  Step 2: Submit update
    Form (POST) → update.php → Validasi → Database (UPDATE WHERE id=X)
                                    ↓
    index.php ← Redirect dengan message ← Success/Error


4. DELETE (Hapus Data)
━━━━━━━━━━━━━━━━━━━━━━━
User Action:
  index.php → Klik tombol "Delete" → Confirm dialog → OK

Data Flow:
  JavaScript confirm() → Yes → delete.php?id=X → Database (DELETE WHERE id=X)
                                    ↓
  index.php ← Redirect dengan message ← Success


5. SEARCH (Cari Data)
━━━━━━━━━━━━━━━━━━━━━━━
User Action:
  index.php → Ketik di search bar → Klik "Cari"

Data Flow:
  Form (GET) → index.php?search=keyword → getAllSepatu($pdo, $search)
                                    ↓
  Database (SELECT WHERE nama_sepatu LIKE %keyword% OR merk LIKE %keyword%)
                                    ↓
  Render filtered results ← Fetch results

1. SQL Injection Prevention
━━━━━━━━━━━━━━━━━━━━━━━━━━━
✓ Prepared Statements dengan PDO
✓ Parameter binding (:placeholder)
✓ No direct SQL concatenation

Example:
  ❌ BAD:  $sql = "SELECT * FROM sepatu WHERE id = $id";
  ✅ GOOD: $stmt = $pdo->prepare("SELECT * FROM sepatu WHERE id = :id");
           $stmt->execute([':id' => $id]);


2. XSS (Cross-Site Scripting) Prevention
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
✓ htmlspecialchars() pada semua output
✓ cleanInput() function untuk sanitasi

Example:
  ❌ BAD:  echo $sepatu['nama_sepatu'];
  ✅ GOOD: echo htmlspecialchars($sepatu['nama_sepatu']);


3. Input Validation
━━━━━━━━━━━━━━━━━━━━━━━━━━━
✓ Required field validation
✓ Data type validation (harga & stok = number)
✓ Positive number validation (>= 0)
✓ Server-side + client-side validation

Example:
  if (!is_numeric($harga) || $harga < 0) {
      // Error handling
  }


4. Error Handling
━━━━━━━━━━━━━━━━━━━━━━━━━━━
✓ Try-catch blocks
✓ PDO::ERRMODE_EXCEPTION
✓ User-friendly error messages
✓ No sensitive info in error messages

1. SELECT ALL (Read)
━━━━━━━━━━━━━━━━━━━━━━━━━━━
SELECT * FROM sepatu ORDER BY id DESC


2. SELECT WITH SEARCH
━━━━━━━━━━━━━━━━━━━━━━━━━━━
SELECT * FROM sepatu 
WHERE nama_sepatu LIKE :search OR merk LIKE :search 
ORDER BY id DESC


3. SELECT BY ID
━━━━━━━━━━━━━━━━━━━━━━━━━━━
SELECT * FROM sepatu WHERE id = :id


4. INSERT (Create)
━━━━━━━━━━━━━━━━━━━━━━━━━━━
INSERT INTO sepatu 
(nama_sepatu, merk, kategori, ukuran, warna, harga, stok, deskripsi) 
VALUES 
(:nama_sepatu, :merk, :kategori, :ukuran, :warna, :harga, :stok, :deskripsi)


5. UPDATE
━━━━━━━━━━━━━━━━━━━━━━━━━━━
UPDATE sepatu SET 
  nama_sepatu = :nama_sepatu,
  merk = :merk,
  kategori = :kategori,
  ukuran = :ukuran,
  warna = :warna,
  harga = :harga,
  stok = :stok,
  deskripsi = :deskripsi
WHERE id = :id


6. DELETE
━━━━━━━━━━━━━━━━━━━━━━━━━━━
DELETE FROM sepatu WHERE id = :id
